    wr(bus, REG_PULL_UP_A, 0xFF)

    wr(bus, REG_INPUT_DISABLE_B, 0x00)
    wr(bus, REG_INPUT_DISABLE_A, 0x00)

    wr(bus, REG_DEBOUNCE_CONFIG, 0b100)

    wr(bus, REG_DEBOUNCE_ENABLE_B, deb_b)
    wr(bus, REG_DEBOUNCE_ENABLE_A, deb_a)

    wr(bus, REG_INT_MASK_B, ~((1<<6)|(1<<3)|(1<<0)|(1<<5)|(1<<4)) & 0xFF)  # enable P8,P11,P14,P12,P13
    wr(bus, REG_INT_MASK_A, ~((1<<0)|(1<<1)|(1<<2)|(1<<3)|(1<<4)|(1<<5)|(1<<6)|(1<<7)) & 0xFF) # P0..P7

    b = rd(bus, REG_DATA_B)
    a = rd(bus, REG_DATA_A)

def decode(enc_idx, a_bit, b_bit, port16):
    a = (port16 >> a_bit) & 1
    b = (port16 >> b_bit) & 1
    state = (a<<1)|b

    last_ab[enc_idx] = state
    return delta  # +1 CW, -1 CCW, 0 none

```python
from smbus2 import SMBus
import RPi.GPIO as GPIO
import time

I2C_BUS = 1
SX = 0x3E         # change if you wired ADDR pins differently
INT_PIN = 17      # Pi BCM pin connected to SX1509 INT

# --- SX1509 registers (subset) ---
REG_INPUT_DISABLE_B = 0x00
REG_INPUT_DISABLE_A = 0x01
REG_PULL_UP_B       = 0x06
REG_PULL_UP_A       = 0x07
REG_DIR_B           = 0x0E
REG_DIR_A           = 0x0F
REG_DATA_B          = 0x10
REG_DATA_A          = 0x11
REG_INT_MASK_B      = 0x12
REG_INT_MASK_A      = 0x13
REG_SENSE_LOW_B     = 0x16
REG_SENSE_HIGH_B    = 0x17
REG_SENSE_LOW_A     = 0x14
REG_SENSE_HIGH_A    = 0x15
REG_INT_SRC_B       = 0x18
REG_INT_SRC_A       = 0x19
REG_CLOCK           = 0x1E     # keep default internal clock
REG_MISC            = 0x1F

# Mapping: [(A,B), ...] bit numbers 0..15  (A/B on same port pair)
PAIRS = [(0,1), (3,4), (6,7), (9,10), (12,13)]  # five encoders, last = tempo

# Helpers
def wr(bus, reg, val): bus.write_byte_data(SX, reg, val & 0xFF)
def rd(bus, reg): return bus.read_byte_data(SX, reg)

def setup_sx1509():
    with SMBus(I2C_BUS) as bus:
        # Inputs on all used pins; others unchanged
        # Port A = P0..P7, Port B = P8..P15
        # Mark all as inputs (safe default)
        wr(bus, REG_DIR_A, 0xFF)
        wr(bus, REG_DIR_B, 0xFF)

        # Enable input buffers (0 = enabled) -> leave 0x00
        wr(bus, REG_INPUT_DISABLE_A, 0x00)
        wr(bus, REG_INPUT_DISABLE_B, 0x00)

        # Pull-ups ON for all pins (A/B float-high, go low on contact)
        wr(bus, REG_PULL_UP_A, 0xFF)
        wr(bus, REG_PULL_UP_B, 0xFF)

        # Interrupt enable (0 = enabled) on just the pins we care about
        mask_a = 0xFF
        mask_b = 0xFF
        for a,b in PAIRS:
            if a <= 7: mask_a &= ~(1<<a)
            else:      mask_b &= ~(1<<(a-8))
            if b <= 7: mask_a &= ~(1<<b)
            else:      mask_b &= ~(1<<(b-8))
        wr(bus, REG_INT_MASK_A, mask_a)
        wr(bus, REG_INT_MASK_B, mask_b)

        # Sense configuration: 2 bits per pin
        # 00 = none, 01 = rising, 10 = falling, 11 = both edges
        def set_both_edge(reg_low, reg_high, pins_0_7):
            low = rd(bus, reg_low)
            high = rd(bus, reg_high)
            for p in pins_0_7:
                # Clear existing two-bit field
                if p < 4:
                    shift = p*2
                    low &= ~(0b11 << shift)
                    low |=  (0b11 << shift)
                else:
                    shift = (p-4)*2
                    high &= ~(0b11 << shift)
                    high |=  (0b11 << shift)
            wr(bus, reg_low, low)
            wr(bus, reg_high, high)

        # Apply both-edge to all A/B pins we use
        pinsA = sorted({p for pair in PAIRS for p in pair if p <= 7})
        pinsB = sorted({p-8 for pair in PAIRS for p in pair if p >= 8})
        set_both_edge(REG_SENSE_LOW_A,  REG_SENSE_HIGH_A,  pinsA)
        set_both_edge(REG_SENSE_LOW_B,  REG_SENSE_HIGH_B,  pinsB)

        # Clear any pending interrupts by dummy reads
        _ = rd(bus, REG_INT_SRC_A); _ = rd(bus, REG_INT_SRC_B)
        _ = rd(bus, REG_DATA_A);    _ = rd(bus, REG_DATA_B)

# Quadrature decoder
last_state = [0]*len(PAIRS)   # store last 2-bit state per encoder

def read_ports(bus):
    a = rd(bus, REG_DATA_A)
    b = rd(bus, REG_DATA_B)
    return a | (b<<8)  # P15..P0 in 16-bit

def decode(enc_idx, a_bit, b_bit, port16):
    a = (port16 >> a_bit) & 1
    b = (port16 >> b_bit) & 1
    state = (a<<1) | b
    prev = last_state[enc_idx]
    # Lookup method: transitions that equal +1/-1
    table = {
        (0,1): +1, (1,3): +1, (3,2): +1, (2,0): +1,
        (0,2): -1, (2,3): -1, (3,1): -1, (1,0): -1
    }
    delta = table.get((prev, state), 0)
    last_state[enc_idx] = state
    return delta

# INT handler
def on_int(_ch):
    # Read and clear sources, then sample the input once
    with SMBus(I2C_BUS) as bus:
        _ = rd(bus, REG_INT_SRC_A); _ = rd(bus, REG_INT_SRC_B)
        ports = read_ports(bus)

    # Process all encoders
    for i,(abit, bbit) in enumerate(PAIRS):
        step = decode(i, abit, bbit, ports)
        if step:
            if i == 4:
                print(f"Tempo step: {step:+d}")
            else:
                print(f"Enc{i+1} step: {step:+d}")

if __name__ == "__main__":
    setup_sx1509()
    # Seed last_state by reading current pins once
    with SMBus(I2C_BUS) as bus:
        p = read_ports(bus)
        for i,(a,b) in enumerate(PAIRS):
            last_state[i] = ((p>>a)&1)<<1 | ((p>>b)&1)

    GPIO.setmode(GPIO.BCM)
    GPIO.setup(INT_PIN, GPIO.IN, pull_up_down=GPIO.PUD_UP)
    GPIO.add_event_detect(INT_PIN, GPIO.FALLING, callback=on_int, bouncetime=1)

    print("SX1509 encoders ready. Rotate knobs…  (Ctrl+C to exit)")
    try:
        while True:
            time.sleep(1)
    except KeyboardInterrupt:
        GPIO.cleanup()
```

from smbus2 import SMBus
import RPi.GPIO as GPIO

I2C_BUS = 1

# --- SX1509 registers (subset) ---
REG_INPUT_DISABLE_B = 0x00
REG_INPUT_DISABLE_A = 0x01
REG_PULL_UP_B       = 0x06
REG_PULL_UP_A       = 0x07
REG_DIR_B           = 0x0E
REG_DIR_A           = 0x0F
REG_DATA_B          = 0x10
REG_DATA_A          = 0x11
REG_INT_MASK_B      = 0x12
REG_INT_MASK_A      = 0x13
REG_SENSE_LOW_B     = 0x16
REG_SENSE_HIGH_B    = 0x17
REG_SENSE_LOW_A     = 0x14
REG_SENSE_HIGH_A    = 0x15
REG_INT_SRC_B       = 0x18
REG_INT_SRC_A       = 0x19

class SX1509Encoders:
    """
    Interrupt-driven SX1509 encoder reader.
    Callbacks:
      on_step(enc_idx:int, delta:int)  # +1 / -1
    """
    def __init__(self, i2c_addr=0x3E, int_pin=17, pairs=None):
        self.addr = i2c_addr
        self.int_pin = int_pin
        self.pairs = pairs or [(0,1),(3,4),(6,7),(9,10),(12,13)]
        self.last_state = [0]*len(self.pairs)
        self.on_step = None

    # --- low-level helpers ---
    def _wr(self, reg, val):
        with SMBus(I2C_BUS) as bus:
            bus.write_byte_data(self.addr, reg, val & 0xFF)

    def _rd(self, reg):
        with SMBus(I2C_BUS) as bus:
            return bus.read_byte_data(self.addr, reg)

    def _read_ports(self):
        a = self._rd(REG_DATA_A)
        b = self._rd(REG_DATA_B)
        return a | (b<<8)

    # --- init ---
    def begin(self):
        # all inputs
        self._wr(REG_DIR_A, 0xFF)
        self._wr(REG_DIR_B, 0xFF)
        # enable input buffers
        self._wr(REG_INPUT_DISABLE_A, 0x00)
        self._wr(REG_INPUT_DISABLE_B, 0x00)
        # pull-ups ON (A/B idle high)
        self._wr(REG_PULL_UP_A, 0xFF)
        self._wr(REG_PULL_UP_B, 0xFF)

        # INT mask (0 = enabled) only on used pins
        mask_a, mask_b = 0xFF, 0xFF
        for a,b in self.pairs:
            (mask_a, mask_b) = (
                mask_a & ~(1<<a) if a<=7 else mask_a,
                mask_b & ~(1<<(a-8)) if a>=8 else mask_b
            )
            (mask_a, mask_b) = (
                mask_a & ~(1<<b) if b<=7 else mask_a,
                mask_b & ~(1<<(b-8)) if b>=8 else mask_b
            )
        self._wr(REG_INT_MASK_A, mask_a)
        self._wr(REG_INT_MASK_B, mask_b)

        # set BOTH edges for all used pins (2 bits per pin)
        def set_both_edge(reg_low, reg_high, pins):
            low  = self._rd(reg_low)
            high = self._rd(reg_high)
            for p in pins:
                if p < 4:
                    s = p*2
                    low  = (low  & ~(0b11<<s)) | (0b11<<s)
                else:
                    s = (p-4)*2
                    high = (high & ~(0b11<<s)) | (0b11<<s)
            self._wr(reg_low, low); self._wr(reg_high, high)

        pinsA = sorted({p for ab in self.pairs for p in ab if p<=7})
        pinsB = sorted({p-8 for ab in self.pairs for p in ab if p>=8})
        set_both_edge(REG_SENSE_LOW_A,  REG_SENSE_HIGH_A,  pinsA)
        set_both_edge(REG_SENSE_LOW_B,  REG_SENSE_HIGH_B,  pinsB)

        # clear ints, seed last_state
        _ = self._rd(REG_INT_SRC_A); _ = self._rd(REG_INT_SRC_B)
        p = self._read_ports()
        for i,(a,b) in enumerate(self.pairs):
            self.last_state[i] = ((p>>a)&1)<<1 | ((p>>b)&1)

        # INT pin on Pi
        GPIO.setmode(GPIO.BCM)
        GPIO.setup(self.int_pin, GPIO.IN, pull_up_down=GPIO.PUD_UP)
        GPIO.add_event_detect(self.int_pin, GPIO.FALLING, callback=self._on_int, bouncetime=1)

    def on_step_callback(self, fn):
        self.on_step = fn

    def _decode(self, i, a_bit, b_bit, port16):
        a = (port16 >> a_bit) & 1
        b = (port16 >> b_bit) & 1
        st  = (a<<1) | b
        prev = self.last_state[i]
        lut = { (0,1):+1,(1,3):+1,(3,2):+1,(2,0):+1,
                (0,2):-1,(2,3):-1,(3,1):-1,(1,0):-1 }
        delta = lut.get((prev,st),0)
        self.last_state[i] = st
        return delta

    def _on_int(self, _ch):
        # latch sources (reading clears), then sample once
        _ = self._rd(REG_INT_SRC_A); _ = self._rd(REG_INT_SRC_B)
        ports = self._read_ports()
        for i,(a,b) in enumerate(self.pairs):
            d = self._decode(i,a,b,ports)
            if d and self.on_step:
                self.on_step(i, d)

    def close(self):
        try:
            GPIO.remove_event_detect(self.int_pin)
        except Exception:
            pass
        GPIO.cleanup(self.int_pin)

class ADS1115Reader:
    """
    Polls ADS1115 channels in a background thread and calls:
      on_sample(channel:int, value:int)  # raw ADC
    You can scale to volts in main if needed (FSR depends on PGA).
    """
    def __init__(self, addr=ADS_ADDR, rate_hz=200, channels=(0,1,2,3)):
        self.addr = addr
        self.rate_hz = rate_hz
        self.channels = channels
        self.on_sample = None
        self._t = None
        self._stop = Event()

    def on_sample_callback(self, fn):
        self.on_sample = fn

    # ---- minimal register ops (single-ended, PGA=±4.096V, DR=250SPS) ----
    REG_CONV = 0x00
    REG_CONF = 0x01

    def _wr16(self, reg, val):
        with SMBus(I2C_BUS) as bus:
            bus.write_i2c_block_data(self.addr, reg, [(val>>8)&0xFF, val&0xFF])

    def _rd16(self, reg):
        with SMBus(I2C_BUS) as bus:
            b = bus.read_i2c_block_data(self.addr, reg, 2)
            return (b[0]<<8) | b[1]

    def _start_single(self, ch):
        mux = {0:0b100,1:0b101,2:0b110,3:0b111}[ch]  # AINx vs GND
        conf =  (1<<15) | (mux<<12) | (0b001<<9) | (0b000<<8) | (0b100<<5) | (1<<8)
        # OS=1 (start), MUX, PGA=±4.096V, MODE=single, DR=250SPS, COMP disabled
        self._wr16(self.REG_CONF, conf)

    def _read_single(self, ch):
        self._start_single(ch)
        time.sleep(1/250.0 + 0.001)  # tiny margin
        v = self._rd16(self.REG_CONV)
        if v & 0x8000: v -= 1<<16  # sign
        return v

    # ---- thread loop ----
    def begin(self):
        if self._t and self._t.is_alive(): return
        self._stop.clear()
        self._t = Thread(target=self._run, daemon=True)
        self._t.start()

    def _run(self):
        period = 1.0 / float(self.rate_hz)
        while not self._stop.is_set():
            t0 = time.time()
            for ch in self.channels:
                val = self._read_single(ch)
                if self.on_sample:
                    self.on_sample(ch, val)
            dt = time.time() - t0
            if dt < period:
                time.sleep(period - dt)

    def close(self):
        self._stop.set()
        if self._t: self._t.join(timeout=1.0)

# A) ADS1115 as “ADC keypad” (one channel, many buttons)

## Wiring (ladder on CH0)
- ADS1115 A0 (CH0) ← node `BTN_ADC`
- `BTN_ADC` → each button → series resistor → **GND**
- `BTN_ADC` → **pull-up** to 3V3 (e.g. 33 kΩ).  
Each button uses a *different* series resistor value, so each press makes a distinct voltage at A0.

Example 5 buttons (values are not critical—just keep them spaced):
| Button | Series R to GND |
|---|---|
| B1 | 1.0 k |
| B2 | 2.2 k |
| B3 | 4.7 k |
| B4 | 10 k |
| B5 | 22 k |

Use 33 k to 3V3 as the common pull-up. Pressing B2 forms a divider: 3V3 * (2.2k / (33k + 2.2k)) ≈ ~0.21 V → a distinct ADC code. We’ll detect by **nearest threshold with tolerance**.

---

# B) ADS1115 per-button (simple, but uses channels)
- Wire each button between **GND** and an ADS channel via a single resistor (e.g. 10 k pull-up to 3V3, button shorts to GND). Read voltage and treat < 0.3 V as pressed. This costs one channel per button, so usually not ideal.

## 1) Minimal button reader on ADS1115 (ladder on CH0)
Save as `midio/ads1115_buttons.py`:

```python
# ads1115_buttons.py
import time
from smbus2 import SMBus

I2C_BUS = 1
ADS_ADDR = 0x48
REG_CONV = 0x00
REG_CONF = 0x01

def _wr16(addr, reg, val):
    with SMBus(I2C_BUS) as bus:
        bus.write_i2c_block_data(addr, reg, [(val>>8)&0xFF, val&0xFF])

def _rd16(addr, reg):
    with SMBus(I2C_BUS) as bus:
        b = bus.read_i2c_block_data(addr, reg, 2)
        return (b[0]<<8) | b[1]

def _start_single(addr, ch):
    mux = {0:0b100,1:0b101,2:0b110,3:0b111}[ch]   # AINx vs GND
    conf = (1<<15) | (mux<<12) | (0b001<<9) | (1<<8) | (0b100<<5)  # OS=1, PGA=±4.096, single, 250SPS
    _wr16(addr, REG_CONF, conf)

def read_raw(addr, ch):
    _start_single(addr, ch)
    time.sleep(1/250.0 + 0.001)
    v = _rd16(addr, REG_CONV)
    if v & 0x8000: v -= 1<<16
    return v

class ADSKeypad:
    """
    One-channel resistor-ladder keypad on ADS1115.
    Provide expected raw targets (codes) for each button, and a tolerance.
    Emits press/release events with simple debounce.
    """
    def __init__(self, addr=ADS_ADDR, channel=0, map_targets=None, tol=1200, debounce_ms=25):
        self.addr = addr
        self.channel = channel
        # map_targets: list of (name, raw_target)
        self.map_targets = map_targets or []
        self.tol = tol
        self.debounce_ms = debounce_ms
        self._last = None
        self._last_change = 0
        self.on_press = None
        self.on_release = None

    def begin(self): pass

    def poll(self):
        raw = read_raw(self.addr, self.channel)
        # find nearest target within tolerance
        hit = None
        best_err = 1e9
        for name, target in self.map_targets:
            err = abs(raw - target)
            if err < best_err and err <= self.tol:
                best_err = err
                hit = name

        now = time.time() * 1000
        if hit != self._last and (now - self._last_change) >= self.debounce_ms:
            # state change
            if hit and self.on_press:   self.on_press(hit)
            if self._last and not hit and self.on_release: self.on_release(self._last)
            self._last = hit
            self._last_change = now

        return raw, hit
```

# midio/encoders_sx1509_poll2.py
# Polling-only SX1509 encoder reader (no GPIO/INT required)

from smbus2 import SMBus
from time import monotonic

I2C_BUS = 1

# SX1509 registers (subset)
REG_INPUT_DISABLE_B = 0x00
REG_INPUT_DISABLE_A = 0x01
REG_PULL_UP_B       = 0x06
REG_PULL_UP_A       = 0x07
REG_DIR_B           = 0x0E
REG_DIR_A           = 0x0F
REG_DATA_B          = 0x10
REG_DATA_A          = 0x11

def _wr8(addr, reg, val):
    with SMBus(I2C_BUS) as bus:
        bus.write_byte_data(addr, reg, val & 0xFF)

def _rd8(addr, reg):
    with SMBus(I2C_BUS) as bus:
        return bus.read_byte_data(addr, reg) & 0xFF

def _read_ports(addr):
    a = _rd8(addr, REG_DATA_A)
    b = _rd8(addr, REG_DATA_B)
    return a | (b << 8)   # 16-bit P15..P0 (bit N corresponds to PN)

class SX1509Encoders:
    """
    Simple polling quadrature decoder for SX1509.
    API:
      begin()
      read_and_zero() -> list[(idx, delta, pressed_bool=False)]
      close()
    """
    def __init__(self, addr=0x3E, pairs=None, poll_hz=500):
        self.addr = addr
        self.pairs = pairs or [(0,1),(3,4),(6,7),(9,10),(12,13)]
        self.poll_hz = poll_hz
        self._last_state = [0]*len(self.pairs)
        self._accum = [0]*len(self.pairs)
        self._t_next = 0.0
        self._period = 1.0/float(self.poll_hz)

    def begin(self):
        # Configure pins as inputs with pull-ups enabled.
        _wr8(self.addr, REG_DIR_A, 0xFF)
        _wr8(self.addr, REG_DIR_B, 0xFF)
        _wr8(self.addr, REG_INPUT_DISABLE_A, 0x00)  # enable input buffers
        _wr8(self.addr, REG_INPUT_DISABLE_B, 0x00)
        _wr8(self.addr, REG_PULL_UP_A, 0xFF)        # pull-ups ON
        _wr8(self.addr, REG_PULL_UP_B, 0xFF)
        # Seed last state
        ports = _read_ports(self.addr)
        for i,(a,b) in enumerate(self.pairs):
            self._last_state[i] = (((ports >> a) & 1) << 1) | ((ports >> b) & 1)
        self._t_next = monotonic()
        return self

    def _sample_once(self):
        ports = _read_ports(self.addr)
        # decode each encoder
        for i,(abit,bbit) in enumerate(self.pairs):
            a = (ports >> abit) & 1
            b = (ports >> bbit) & 1
            st  = (a<<1) | b
            prev = self._last_state[i]
            # quadrature LUT (+1 / -1 / 0)
            lut = {
                (0,1):+1,(1,3):+1,(3,2):+1,(2,0):+1,
                (0,2):-1,(2,3):-1,(3,1):-1,(1,0):-1
            }
            d = lut.get((prev, st), 0)
            if d:
                self._accum[i] += d
            self._last_state[i] = st

    def read_and_zero(self):
        """
        Poll at up to poll_hz. Returns list of (idx, delta, False)
        since we're not reading push switches here.
        """
        out = []
        now = monotonic()
        # Ensure at least one sample per call; if loop runs faster, rate-limit
        if now >= self._t_next:
            self._sample_once()
            self._t_next = now + self._period

        for i,acc in enumerate(self._accum):
            if acc:
                out.append((i, acc, False))
                self._accum[i] = 0
        return out

    def close(self):
        # nothing persistent to tear down in polling mode
        pass

## Assistant · 2025-11-04 23:15

Short answer:

- **No special LED “protocol.”** That RGB is just 3 LED dies (R, G, B).  
- We’ll drive each color from **three SX1509 pins**.  
- **Brightness = PWM**, generated **inside the SX1509**.  
- We talk to the SX1509 over **I²C** (100/400 kHz). The SX1509 does the PWM; the LED itself has no protocol like WS2812.

### How to wire it (matches your board silkscreen)
- **R → SX1509 P3**, **G → SX1509 P4**, **B → SX1509 P5** (any 3 pins are fine; these are just an example)
- **GND (LED cathode)** → **GND**
- Encoder: **A→P0**, **B→P1**, **C→GND** (as you already have)
- Switch: **SW→P2** (pull-up + debounce ON), **G (switch common)** → **GND**

> Your PCB labels “R G B GND SW G” imply a **common-cathode** RGB: GND is the common pin; drive R/G/B pins **HIGH** to light.

### Current limiting
Make sure there are **series resistors** on R/G/B (typically 150–330 Ω). If that little board doesn’t already have them, add them in-line between each SX1509 pin and the LED pin.

### What the SX1509 does for the RGB
- Each pin can be put into **LED driver (PWM) mode** with ~8-bit duty (0–255).  
- You set duty per color over **I²C**, e.g. `(R,G,B) = (0..255)`.  
- It also has simple **fade/ramp** hardware (nice for breath/pulse without CPU load).

### Mental model
- **Protocol:** I²C only (to the SX1509).  
- **PWM:** Generated by SX1509; the RGB LED is “dumb” 3-wire (plus GND).  
- **Logic level:** 3.3 V.  
- **Polarity:** With common-cathode, **higher duty = brighter** (active-HIGH). If you ever use a common-ANODE LED, it’s the opposite (active-LOW).

### Quick map you can lock in for SUPERCORE
- Enc A/B: **(P0, P1)**  
- Enc SW: **P2** (debounced)  
- RGB: **P3=P_RED, P4=P_GREEN, P5=P_BLUE**  
- Side buttons later: **P6, P7** (debounced)
